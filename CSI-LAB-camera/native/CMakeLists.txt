cmake_minimum_required(VERSION 3.16)
project(csiVideoCapture LANGUAGES CXX)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs videoio)
find_package(OpenSSL REQUIRED)
add_executable(cameraCapture cameraCapture.cpp)
target_compile_features(cameraCapture PRIVATE cxx_std_17)
target_compile_options(cameraCapture PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(cameraCapture PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(cameraCapture PRIVATE ${OpenCV_LIBS} OpenSSL::Crypto)
